import hashlib
import json
import subprocess
from pathlib import Path

from tools.glm_perf.resident_harness import ResidentClient
from tools.glm_perf.resident_native import NativeManifest
from tools.qwen4exp.thermal_controller import parse_temperatures

ROOT = Path("/srv/ai/src/qwen-cast-layout-20261010")
RUNTIME = Path("/srv/ai/src/qwen-performance-paced-tp6-20261010")
client = ResidentClient("http://127.0.0.1:8001", 6, 180)
temps = parse_temperatures(subprocess.check_output(["npu-smi", "info"], text=True, timeout=5), 6)
assert max(temps) < 85
assert not client.request("/is_paused", method="GET")["is_paused"]
manifest = json.loads(Path("/home/matteius/experiments/qwen38-decode-next-20261005/native-hc-residual/native-v1.json").read_text())
manifest["assets"] = manifest["assets"][:1]
for name in ("direct_hc_residual.py", "resident_candidates/fused_hc_projection.py", "resident_candidates/fused_hc_residual.py"):
    path = RUNTIME / "tools/qwen4exp" / name
    manifest["assets"].append({"path": str(path), "sha256": hashlib.sha256(path.read_bytes()).hexdigest()})
manifest["validation_source"] += '''
original_validate = validate
def validate(op):
    import torch
    report = original_validate(op)
    with torch.inference_mode():
        for rows in (3, 18, 128, 640, 2560):
            gen = torch.Generator().manual_seed(20261010 + rows)
            hyper = torch.randn(rows, 10240, generator=gen).half().npu()
            block = torch.randn(rows, 2560, generator=gen).half().npu()
            for dtype in (torch.float16, torch.float32):
                injection = (torch.rand(rows, 4, generator=gen) * 2).to(dtype).npu()
                expected = (hyper.float().view(rows, 4, 2560) + block.float()[:,None] * injection.float()[:,:,None]).flatten(1).half()
                actual = op(hyper, block, injection)
                torch.testing.assert_close(actual.cpu(), expected.cpu(), rtol=0, atol=0)
    report["geometry_cases"] = 10
    report["rows"] = [3, 18, 128, 640, 2560]
    return report
'''
(ROOT / "hc-native-manifest.json").write_text(json.dumps(manifest, indent=2) + "\n")
loaded = client.load_native(NativeManifest(manifest))
(ROOT / "hc-native-loaded.json").write_text(json.dumps(loaded, indent=2) + "\n")
print(json.dumps({"loaded": len(loaded), "temperatures_before": temps}), flush=True)
