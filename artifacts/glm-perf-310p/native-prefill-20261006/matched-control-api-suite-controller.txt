import sys,json,time
from dataclasses import replace
from pathlib import Path
sys.path.insert(0,'/srv/ai/src/glm-selective-w3-nz-test-20261004')
from tools.glm_perf.resident_harness import ResidentClient
from tools.glm_perf.suite import make_groups,run_groups,summarize
root=Path('/home/matteius/experiments/glm-reconstruction-20261006')
ready=Path('/srv/ai/artifacts/glm-native-prefill-20261006/trace-v45/export-status.json')
deadline=time.monotonic()+300
while not ready.exists():
 if time.monotonic()>deadline: raise TimeoutError('prefill export is not complete')
 time.sleep(2)
status=json.loads(ready.read_text())
assert len(status)==4 and all(r['exit_code']==0 for r in status)
client=ResidentClient('http://127.0.0.1:8001',timeout=900)
from tools.glm_perf.resident_control import Control
import uuid
source=(root/'native-v41-bf16-v5-candidate.py').read_text()
client.switch(Control(uuid.uuid4().hex,candidate='native_v41_current_control',source=source))
before=client._acknowledged_rpc('resident_status',matches=client._status_matches)
assert all(r['candidate']=='native_v41_current_control' and not r['graphs_dirty'] and not r['native_failed'] for r in before)
result={'version':41,'indexer_version':5,'before':before,'complete':False,'records':[]}
short=[(n,[replace(c,max_tokens=64) for c in cs]) for n,cs in make_groups(['short'],[])]
quality=[c for _,cs in make_groups(['quality'],[]) for c in cs]
groups=list(short)
groups.extend(('quality_4_'+str(i//4),[replace(c,max_tokens=128) for c in quality[i:i+4]]) for i in range(0,len(quality),4))
groups.extend(make_groups(['tool'],[]))
for name,cases in groups:
 rows=run_groups([(name,cases)],client.base_url,'glm53-flash-selective-w3',42,timeout_s=900)
 result['records'].extend(rows);result['summary']=summarize(result['records'])
 (root/'native-v41-current-control-trial.json').write_text(json.dumps(result,indent=2))
 print(name,json.dumps(summarize(rows)),flush=True)
 assert all(r['valid'] for r in rows)
result['after_requests']=client._acknowledged_rpc('resident_status',matches=client._status_matches)
assert sorted((r['pid'],r['weight_storage_digest']) for r in before)==sorted((r['pid'],r['weight_storage_digest']) for r in result['after_requests'])
assert all(not r['native_failed'] and not r['graphs_dirty'] and r['reconstruction']['fallback_dispatches']==0 for r in result['after_requests'])
result['complete']=True
(root/'native-v41-current-control-trial.json').write_text(json.dumps(result,indent=2))
print('complete',json.dumps(result['summary']),flush=True)

client.switch(Control(uuid.uuid4().hex,candidate='native_v50_bf16_v5',source=(root/'native-v50-bf16-v5-candidate.py').read_text()))
