cmake_minimum_required(VERSION 3.14)
set(CMAKE_CXX_COMPILER "${TOOLCHAIN}")
project(PleArmAccess LANGUAGES CXX)
if(NOT RESOURCE_TYPE STREQUAL "Ascend")
    message(FATAL_ERROR "This diagnostic must execute on the device ARM CPU")
endif()
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${PROJECT_BINARY_DIR}/${RELEASE_DIR}")
add_library(${UDF_TARGET_LIB} SHARED access.cpp)
target_compile_features(${UDF_TARGET_LIB} PRIVATE cxx_std_17)
target_compile_options(${UDF_TARGET_LIB} PRIVATE -O2 -Wall -Wextra -Werror)
target_compile_definitions(${UDF_TARGET_LIB} PRIVATE _GLIBCXX_USE_CXX11_ABI=0)
target_include_directories(${UDF_TARGET_LIB} PRIVATE "$ENV{ASCEND_HOME_PATH}/include")
target_link_libraries(${UDF_TARGET_LIB} PRIVATE "$ENV{ASCEND_HOME_PATH}/devlib/device/libflow_func.so")
