import fcntl,json,sys,uuid
from pathlib import Path
from tools.glm_perf.resident_control import Control
from tools.glm_perf.resident_harness import ResidentClient
c=ResidentClient('http://127.0.0.1:8000',6,timeout=3600)
with Path('/tmp/minimax-m31-resident.lock').open('w') as lock:
 fcntl.flock(lock,fcntl.LOCK_EX|fcntl.LOCK_NB)
 assert not c.request('/is_paused',method='GET')['is_paused']
 assert not c.request('/resident/maintenance',method='GET')['maintenance']
 before=c.rpc('resident_status');c.request('/resident/maintenance')
 try:
  source=Path('/workspace/m31-combined-hooks.py').read_text()
  c.switch(Control(uuid.uuid4().hex,'direct-target','combined_kilo',source));c.resume()
  c.request('/resident/maintenance',method='DELETE')
  print('READY',flush=True)
  sys.stdin.readline()
 finally:
  c.request('/resident/maintenance')
  c.switch(Control(uuid.uuid4().hex,'direct-target','baseline',''));c.resume()
  c.request('/resident/maintenance',method='DELETE')
 after=c.rpc('resident_status')
 stable=[(x['rank'],x['pid'],x['weight_storage_digest']) for x in before]==[(x['rank'],x['pid'],x['weight_storage_digest']) for x in after]
 Path('/workspace/m31-combined-kilo-receipt.json').write_text(json.dumps({'stable_workers_and_weights':stable,'before':before,'after':after},indent=2))
 assert stable
 print('RESTORED',flush=True)
