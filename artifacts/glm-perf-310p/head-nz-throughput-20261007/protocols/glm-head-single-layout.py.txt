import gc
import hashlib
import json
from pathlib import Path
import torch
import torch_npu


def prepare_single_head_layout(native_resources):
    from vllm_ascend._310p.worker_310p import NPUWorker310
    from tools.glm_perf import resident_worker as rw
    key='_single_nz_head_layout_20261007_v1'
    workers=[obj for obj in gc.get_objects() if isinstance(obj,NPUWorker310)]
    if len(workers)!=1:
        raise ValueError('expected one resident worker')
    worker=workers[0]
    if key not in native_resources:
        heads=[obj for obj in gc.get_objects() if type(obj).__name__=='AscendParallelLMHead310']
        if len(heads)!=1:
            raise ValueError('expected one shared target/draft head')
        head=heads[0]
        if tuple(head.weight.shape)!=(38720,4096) or head.weight.dtype!=torch.float16 or torch_npu.get_npu_format(head.weight)!=2:
            raise ValueError('unexpected original head layout')
        def weight_records():
            rows=[]
            for i,wrapper in enumerate(worker._resident_wrappers()):
                for name,p in rw.resident_weight_tensors(wrapper.runnable):
                    rows.append((i,name,p.untyped_storage().data_ptr(),tuple(p.shape),str(p.dtype),tuple(p.stride()),p.storage_offset(),p is head.weight))
            return rows
        before=weight_records()
        # Back up logical bytes on CPU before any storage mutation. Old graphs
        # are released while the server is explicitly paused and drained.
        backup=head.weight.detach().cpu()
        state=dict(head=head,backup=backup,complete=False)
        native_resources[key]=state
        worker._resident_session().graphs_dirty=True
        torch.npu.synchronize()
        rw.clear_graph_state(worker._resident_wrappers(),[rw.get_graph_params(),rw.get_draft_graph_params(),rw.get_draft_graph_prefill_params()])
        gc.collect();torch.npu.empty_cache()
        packed=torch_npu.npu_format_cast(head.weight.data,29)
        head.weight.data=packed
        del packed
        torch.npu.synchronize();torch.npu.empty_cache()
        actual=head.weight.detach().cpu()
        assert torch.equal(actual,backup),'head byte preservation failed'
        digest=hashlib.sha256(backup.numpy().tobytes()).hexdigest()
        del actual
        after=weight_records()
        assert [r for r in before if not r[-1]]==[r for r in after if not r[-1]],'non-head weight storage changed'
        assert [(r[0],r[1],r[3:]) for r in before if r[-1]]==[(r[0],r[1],r[3:]) for r in after if r[-1]],'head logical metadata changed'
        state['complete']=True
        report=dict(complete=True,original_format=2,packed_format=torch_npu.get_npu_format(head.weight),sha256=digest,logical_bytes_preserved=True,non_head_storage_unchanged=True,head_storage_records_before=[r for r in before if r[-1]],head_storage_records_after=[r for r in after if r[-1]],device_head_copies=1,cpu_backup_bytes=backup.numel()*backup.element_size())
        state['report']=report
        Path('/home/matteius/experiments/glm-reconstruction-20261006/single-nz-head-rank'+str(torch.distributed.get_rank())+'.json').write_text(json.dumps(report,indent=2))
    if not native_resources[key]['complete']:
        raise RuntimeError('incomplete layout transaction; restore CPU backup first')
    return {'vllm_ascend._310p.worker_310p:NPUWorker310.resident_status':NPUWorker310.resident_status}


def replacements(native_resources):
    return prepare_single_head_layout(native_resources)
