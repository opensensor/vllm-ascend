import hashlib,json,subprocess,uuid
from pathlib import Path
from tools.glm_perf.resident_control import Control
from tools.glm_perf.resident_harness import ResidentClient
from tools.glm_perf.resident_native import NativeManifest
from tools.qwen4exp.thermal_controller import parse_temperatures
root=Path('/srv/ai/src/qwen-performance-evidence-20261010')
runtime=Path('/srv/ai/src/qwen-performance-mtp-tp6-v2-20261010-8fce905')
client=ResidentClient('http://127.0.0.1:8001',6,300)
temps=parse_temperatures(subprocess.check_output(['npu-smi','info'],text=True,timeout=5),6)
assert max(temps)<85
assert client.request('/is_paused',method='GET')['is_paused'] is False
library=root/'diagnostic-marker-v14.so'
assets=[runtime/'tools/qwen4exp/resident_candidates'/name for name in ['fused_hc_projection.py','fused_hc_residual.py']]
manifest=NativeManifest({'name':'qwen_stage_diagnostic_v14','libraries':[{'path':str(library),'sha256':hashlib.sha256(library.read_bytes()).hexdigest()}],
 'assets':[{'path':str(p),'sha256':hashlib.sha256(p.read_bytes()).hexdigest()} for p in assets],
 'operators':['qwen_stage_diagnostic_v14::marker'],'validation_source':(root/'fused-hc-validation-v14.py').read_text()})
(root/'mtp-v2-fused-hc-manifest.json').write_text(manifest.payload)
client.load_native(manifest)
client.request('/pause?mode=wait&clear_cache=true')
try:
 receipts=client.rpc('resident_fused_hc_gate_v14')
 (root/'mtp-v2-fused-hc-math.json').write_text(json.dumps(receipts,indent=2))
 assert all(v['passed'] for v in receipts),receipts
 source=(runtime/'tools/qwen4exp/resident_candidates/fused_hc_residual.py').read_text()
 result=client.switch(Control(uuid.uuid4().hex,candidate='fused_native_hc',source=source,recapture=True))
 (root/'mtp-v2-fused-hc-selected-v2.json').write_text(json.dumps(result,indent=2))
except Exception:
 # The cleanup candidate releases joined weights only after old graphs clear.
 source=(runtime/'tools/qwen4exp/resident_candidates/release_hc_projection.py').read_text()
 cleanup=client.switch(Control(uuid.uuid4().hex,candidate='release_hc_projection',source=source,recapture=True))
 result=client.switch(Control(uuid.uuid4().hex,recapture=True))
 (root/'mtp-v2-fused-hc-recovery-v2.json').write_text(json.dumps(result,indent=2))
 client.resume()
 raise
client.resume()
print(json.dumps({'candidate':'fused_native_hc','cases_per_rank':[v['cases'] for v in receipts],'temperatures_before':temps}),flush=True)
