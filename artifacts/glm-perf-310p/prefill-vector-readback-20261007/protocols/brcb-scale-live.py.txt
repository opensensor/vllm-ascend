def idle():

 values=[float(value) for value in re.findall(r'^vllm:num_requests_(?:running|waiting)(?:\{[^\n]*\})?\s+([0-9.eE+-]+)$',metrics,re.M)]
 assert values and not any(values),'external request running; stop comparisons'
 return metrics
