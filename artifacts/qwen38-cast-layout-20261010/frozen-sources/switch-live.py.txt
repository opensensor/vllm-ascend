import argparse
import json
import subprocess
import urllib.request
import uuid
from pathlib import Path

from tools.glm_perf.resident_control import Control
from tools.glm_perf.resident_harness import ResidentClient
from tools.qwen4exp.thermal_controller import parse_temperatures

ROOT = Path("/srv/ai/src/qwen-cast-layout-20261010")
parser = argparse.ArgumentParser()
parser.add_argument("arm")
parser.add_argument("source", nargs="?")
args = parser.parse_args()
output = ROOT / (args.arm + "-selected.json")
assert not output.exists()
client = ResidentClient("http://127.0.0.1:8001", 6, 180)
temperatures = parse_temperatures(subprocess.check_output(["npu-smi", "info"], text=True, timeout=5), 6)
assert max(temperatures) < 85
assert not client.request("/is_paused", method="GET")["is_paused"]
metrics = urllib.request.urlopen(client.base_url + "/metrics", timeout=10).read().decode()
assert all(float(line.rsplit(" ", 1)[1]) == 0 for line in metrics.splitlines()
           if line.startswith(("vllm:num_requests_running{", "vllm:num_requests_waiting{")))
source = (ROOT / args.source).read_text() if args.source else ""
result = client.switch(Control(uuid.uuid4().hex, candidate=args.arm if source else "baseline",
                               source=source, recapture=True))
output.write_text(json.dumps({"workers": result, "temperatures_before": temperatures}, indent=2) + "\n")
print(json.dumps({"arm": args.arm, "ranks": len(result), "before_c": temperatures}), flush=True)
