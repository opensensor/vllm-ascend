def validate(resources):

 def reference(x,res,p,comb):

 for dtype in (torch.float16,torch.float32):
  for rows in (1,2,8):
   x=torch.randn(rows,4096,generator=generator,dtype=dtype).npu()
   r=torch.randn(rows,4,4096,generator=generator).half().float().npu()
   p=torch.randn(rows,4,1,generator=generator).sigmoid().half().float().npu()
   comb=torch.randn(rows,4,4,generator=generator).softmax(-1).half().float().npu()
   torch.testing.assert_close(native(x,r,p,comb).cpu(),reference(x,r,p,comb).cpu(),rtol=0,atol=0)
   graph=torch.npu.NPUGraph()

   x.mul_(1.125);r.copy_((r*0.875).half().float());p.copy_((p*0.75).half().float());comb.copy_(comb.flip(-1).contiguous())
   graph.replay();torch.npu.synchronize()
   torch.testing.assert_close(result.cpu(),reference(x,r,p,comb).cpu(),rtol=0,atol=0)
   cases+=1
 x=torch.full((1,4096),1+2**-11+2**-22,dtype=torch.float32,device='npu')
