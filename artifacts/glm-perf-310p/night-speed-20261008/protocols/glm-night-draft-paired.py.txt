def metrics():
 with urllib.request.urlopen(c.base_url+'/metrics',timeout=30) as response:text=response.read().decode()
 return {k:float(v) for k,v in re.findall(r'^(vllm:spec_decode_num_(?:drafts|draft_tokens|accepted_tokens)_total)(?:\{[^\n]*\})?\s+([0-9.eE+-]+)$',text,re.M)}
def stream(prompt,generated,label):
 req=urllib.request.Request(c.base_url+'/v1/completions',data=json.dumps({'model':'glm53-flash-selective-w3','prompt':prompt,'max_tokens':generated,'temperature':0,'ignore_eos':True,'stream':True,'stream_options':{'include_usage':True}}).encode(),headers={'Content-Type':'application/json'})
 began=time.perf_counter();first=None;last=None;usage=None;fragments=[];done=False
 with urllib.request.urlopen(req,timeout=900) as response:
  for raw in response:
   if not raw.startswith(b'data:'):continue
   data=raw[5:].strip()
   if data==b'[DONE]':done=True;break
   event=json.loads(data);assert 'error' not in event
   if event.get('usage'):usage=event['usage']
   for choice in event.get('choices',[]):
    if choice.get('text'):
     now=time.perf_counter();first=now if first is None else first;last=now;fragments.append(choice['text'])
 assert usage and usage['completion_tokens']==generated and done
 r=dict(label=label,usage=usage,total_s=time.perf_counter()-began,ttft_s=first-began,generation_tok_s=(generated-1)/(last-first) if generated>1 else None,text=''.join(fragments))
 records.append(r);print(json.dumps(r),flush=True)
 (root/'night-speed-20261008/draft-paired-serving.json').write_text(json.dumps(dict(complete=False,records=records),indent=2))
