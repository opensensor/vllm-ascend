from pathlib import Path
import tarfile,json,hashlib
root=Path('/srv/ai/artifacts/glm-prefill-weight-reuse-20261007')
output=root/'night-speed-20261008/qualified-kda-beta.tar.gz'
parent=root/'night-kda-beta-vector-full'
paths=list((parent/'opp-beta-vector').rglob('*'))
paths += [parent/'build.json',parent/'stage.json',parent/'validate-full-kda.py']
paths += list((root/'build-kda-beta-vector-v978').glob('*'))
records={}
with tarfile.open(output,'w:gz') as archive:
 for path in paths:
  if path.is_file() and path.suffix!='.pyc':
   archive.add(path,arcname=str(path.relative_to(root)),recursive=False)
   records[str(path.relative_to(root))]=hashlib.sha256(path.read_bytes()).hexdigest()
(root/'night-speed-20261008/qualified-kda-beta-assets.json').write_text(json.dumps(dict(archive=str(output),sha256=hashlib.sha256(output.read_bytes()).hexdigest(),assets=records),indent=2)+'\n')
print(output.stat().st_size,flush=True)
