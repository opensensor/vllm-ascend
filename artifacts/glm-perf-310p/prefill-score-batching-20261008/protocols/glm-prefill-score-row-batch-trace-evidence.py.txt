import gzip,tarfile,hashlib,json
from pathlib import Path
root=Path('/srv/ai/artifacts/glm-prefill-weight-reuse-20261007/night-speed-20261008/trace-kda-score-row-batch')
out=root/'delivery';out.mkdir(exist_ok=False)
for label in ['decode_c1','prefill_1280']:

 (out/(label+'-host-steps.json')).write_bytes((rank/'host-steps.json').read_bytes())
