def idle():
 with urllib.request.urlopen(client.base_url+"/metrics",timeout=30) as reply:metrics=reply.read().decode()
 counts=[float(v) for v in re.findall(r'^vllm:num_requests_(?:running|waiting)(?:\{[^\n]*\})?\s+([0-9.eE+-]+)$',metrics,re.M)]
 assert counts and not any(counts),"GLM user request active; defer benchmark"
def stream(tokens,output_tokens,nonce):
 prompt=[101+nonce]+[43]*(tokens-2)+[201+nonce]
 payload={'model':'glm53-flash-selective-w3','prompt':prompt,'max_tokens':output_tokens,'temperature':0,'ignore_eos':True,'stream':True,'stream_options':{'include_usage':True}}
 request=urllib.request.Request(client.base_url+'/v1/completions',data=json.dumps(payload).encode(),headers={'Content-Type':'application/json'})
 start=time.monotonic();first=None;last=None;usage=None;done=False;fragments=[]
 with urllib.request.urlopen(request,timeout=900) as reply:
  for raw in reply:
   if not raw.startswith(b'data:'):continue
   data=raw[5:].strip()
   if data==b'[DONE]':done=True;break
   event=json.loads(data);assert 'error' not in event,event
   if event.get('usage'):usage=event['usage']
   for choice in event.get('choices',[]):
    if choice.get('text'):
     now=time.monotonic();first=now if first is None else first;last=now;fragments.append(choice['text'])
 assert done and usage and first is not None
 assert usage['prompt_tokens']==tokens and usage['completion_tokens']==output_tokens
 elapsed=time.monotonic()-start
 return dict(prompt_tokens=tokens,completion_tokens=output_tokens,ttft_s=first-start,total_s=elapsed,
             generation_tok_s=(output_tokens-1)/(elapsed-(first-start)) if output_tokens>1 else None,
             visible_generation_tok_s=(output_tokens-1)/(last-first) if output_tokens>1 and last>first else None,
             text=''.join(fragments),nonce=nonce)
