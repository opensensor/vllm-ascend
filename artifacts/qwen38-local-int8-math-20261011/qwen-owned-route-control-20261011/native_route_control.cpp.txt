// Small stable dispatch and owned ordered reduction; no host route-count reads.
#include "kernel_operator.h"
namespace qwen_route_control {
using namespace AscendC;
constexpr uint32_t MAX_ROUTES = 60, MAX_EXPERTS = 128, MAX_TOP_K = 10, COLUMN_TILE = 256, LANES = 8;
class FixedDispatch {
 public:
  __aicore__ inline void Run(GM_ADDR ids, GM_ADDR tokens, GM_ADDR inverse, GM_ADDR ends, GM_ADDR descriptor) {
    auto config = reinterpret_cast<__gm__ int64_t*>(descriptor);
    const int64_t rows = config[0], topk = config[1], experts = config[2], offset = config[3], routes = rows * topk;
    ASCENDC_ASSERT(rows > 0 && routes <= MAX_ROUTES && topk > 0 && topk <= MAX_TOP_K && experts > 0 &&
                       experts <= MAX_EXPERTS && offset >= 0,
                   { KERNEL_LOG(KERNEL_ERROR, "invalid fixed route geometry"); });
    GlobalTensor<int64_t> routeIds, groupEnds;
    GlobalTensor<int32_t> sortedTokens, inverseOrder;
    routeIds.SetGlobalBuffer(reinterpret_cast<__gm__ int64_t*>(ids));
    groupEnds.SetGlobalBuffer(reinterpret_cast<__gm__ int64_t*>(ends));
    sortedTokens.SetGlobalBuffer(reinterpret_cast<__gm__ int32_t*>(tokens));
    inverseOrder.SetGlobalBuffer(reinterpret_cast<__gm__ int32_t*>(inverse));
    TPipe pipe;
    TBuf<TPosition::VECCALC> storage;
    pipe.InitBuffer(storage, 2 * MAX_ROUTES * sizeof(int32_t));
    auto keys = storage.Get<int32_t>();
    auto order = keys[MAX_ROUTES];
    for (int64_t route = 0; route < routes; ++route) {
      const int64_t id = routeIds.GetValue(route) - offset;
      const int32_t key = id >= 0 && id < experts ? id : experts;
      int64_t insertion = route;
      while (insertion > 0 && keys.GetValue(insertion - 1) > key) {
        keys.SetValue(insertion, keys.GetValue(insertion - 1));
        order.SetValue(insertion, order.GetValue(insertion - 1));
        --insertion;
      }
      keys.SetValue(insertion, key);
      order.SetValue(insertion, route);
    }
    for (int64_t route = 0; route < routes; ++route) {
      const int32_t original = order.GetValue(route);
      sortedTokens.SetValue(route, original / topk);
      inverseOrder.SetValue(original, route);
    }
    int64_t end = 0;
    for (int64_t expert = 0; expert < experts; ++expert) {
      while (end < routes && keys.GetValue(end) <= expert) ++end;
      groupEnds.SetValue(expert, end);
    }
    PipeBarrier<PIPE_ALL>();
    pipe.Destroy();
  }
};

template <bool HALF_PRODUCT = false, bool FP32_BOUNDARY = false>
class OwnedFinalize {
 public:
  __aicore__ inline void Run(GM_ADDR input, GM_ADDR weights, GM_ADDR inverse, GM_ADDR ends, GM_ADDR output,
                             GM_ADDR descriptor) {
    auto config = reinterpret_cast<__gm__ int64_t*>(descriptor);
    const int64_t tokens = config[0], topk = config[1], width = config[2], experts = config[3], routes = tokens * topk;
    ASCENDC_ASSERT(tokens > 0 && routes <= 25600 && topk > 0 && topk <= MAX_TOP_K && width > 0 &&
                       width % COLUMN_TILE == 0 && experts > 0 && experts <= MAX_EXPERTS,
                   { KERNEL_LOG(KERNEL_ERROR, "invalid owned finalize geometry"); });
    GlobalTensor<half> routed;
    GlobalTensor<float> scales, result;
    GlobalTensor<int32_t> indices;
    GlobalTensor<int64_t> groupEnds;
    routed.SetGlobalBuffer(reinterpret_cast<__gm__ half*>(input));
    scales.SetGlobalBuffer(reinterpret_cast<__gm__ float*>(weights));
    indices.SetGlobalBuffer(reinterpret_cast<__gm__ int32_t*>(inverse));
    groupEnds.SetGlobalBuffer(reinterpret_cast<__gm__ int64_t*>(ends));
    result.SetGlobalBuffer(reinterpret_cast<__gm__ float*>(output));
    const int64_t localEnd = groupEnds.GetValue(experts - 1), tiles = width / COLUMN_TILE;
    ASCENDC_ASSERT(localEnd >= 0 && localEnd <= routes, { KERNEL_LOG(KERNEL_ERROR, "invalid local route boundary"); });
    TPipe pipe;
    TBuf<TPosition::VECCALC> storage;
    pipe.InitBuffer(storage, COLUMN_TILE * (sizeof(half) + 2 * sizeof(float)) + LANES * (sizeof(float) + sizeof(half)));
    auto halves = storage.Get<half>();
    auto values = storage.Get<float>()[COLUMN_TILE / 2];
    auto accumulator = values[COLUMN_TILE];
    auto scale = accumulator[COLUMN_TILE];
    auto roundedScale = scale[LANES].ReinterpretCast<half>();
    for (int64_t task = GetBlockIdx(); task < tokens * tiles; task += GetBlockNum()) {
      const int64_t token = task / tiles, column = task % tiles * COLUMN_TILE;
      Duplicate(accumulator, 0.0f, COLUMN_TILE);
      PipeBarrier<PIPE_V>();
      for (int64_t route = 0; route < topk; ++route) {
        const int64_t row = indices.GetValue(token * topk + route);
        ASCENDC_ASSERT(row >= 0 && row < routes, { KERNEL_LOG(KERNEL_ERROR, "invalid inverse route"); });
        if (row >= localEnd) continue;
        SetFlag<HardEvent::V_MTE2>(EVENT_ID0);
        WaitFlag<HardEvent::V_MTE2>(EVENT_ID0);
        DataCopy(halves, routed[row * width + column], COLUMN_TILE);
        SetFlag<HardEvent::MTE2_V>(EVENT_ID0);
        WaitFlag<HardEvent::MTE2_V>(EVENT_ID0);
        Cast(values, halves, RoundMode::CAST_NONE, COLUMN_TILE);
        Duplicate(scale, scales.GetValue(token * topk + route), LANES);
        PipeBarrier<PIPE_V>();
        if constexpr (!FP32_BOUNDARY) {
          Cast(roundedScale, scale, RoundMode::CAST_NONE, LANES);
          PipeBarrier<PIPE_V>();
          Cast(scale, roundedScale, RoundMode::CAST_NONE, LANES);
          PipeBarrier<PIPE_V>();
        }
        Mul(values, values, scale, 64, COLUMN_TILE / 64, {1, 1, 0, 8, 8, 0});
        PipeBarrier<PIPE_V>();
        if constexpr (HALF_PRODUCT) {
          Cast(halves, values, RoundMode::CAST_NONE, COLUMN_TILE);
          PipeBarrier<PIPE_V>();
          Cast(values, halves, RoundMode::CAST_NONE, COLUMN_TILE);
          PipeBarrier<PIPE_V>();
        }
        Add(accumulator, accumulator, values, COLUMN_TILE);
        PipeBarrier<PIPE_V>();
      }
      if constexpr (!FP32_BOUNDARY) {
        Cast(halves, accumulator, RoundMode::CAST_NONE, COLUMN_TILE);
        PipeBarrier<PIPE_V>();
        Cast(accumulator, halves, RoundMode::CAST_NONE, COLUMN_TILE);
        PipeBarrier<PIPE_V>();
      }
      SetFlag<HardEvent::V_MTE3>(EVENT_ID0);
      WaitFlag<HardEvent::V_MTE3>(EVENT_ID0);
      DataCopy(result[token * width + column], accumulator, COLUMN_TILE);
      SetFlag<HardEvent::MTE3_V>(EVENT_ID0);
      WaitFlag<HardEvent::MTE3_V>(EVENT_ID0);
    }
    pipe.Destroy();
  }
};
}  // namespace qwen_route_control
extern "C" __global__ __aicore__ void qwen_fixed_route_dispatch_v1(GM_ADDR ids, GM_ADDR tokens, GM_ADDR inverse,
                                                                   GM_ADDR ends, GM_ADDR descriptor) {
  KERNEL_TASK_TYPE_DEFAULT(KERNEL_TYPE_AICORE);
  AscendC::InitSocState();
  qwen_route_control::FixedDispatch operation;
  operation.Run(ids, tokens, inverse, ends, descriptor);
}
#define DEFINE_FINALIZE(SYMBOL, HALF_PRODUCT, FP32_BOUNDARY)                                                  \
  extern "C" __global__ __aicore__ void SYMBOL(GM_ADDR input, GM_ADDR weights, GM_ADDR inverse, GM_ADDR ends,   \
                                               GM_ADDR output, GM_ADDR descriptor) {                          \
    KERNEL_TASK_TYPE_DEFAULT(KERNEL_TYPE_AICORE);                                                              \
    AscendC::InitSocState();                                                                                   \
    qwen_route_control::OwnedFinalize<HALF_PRODUCT, FP32_BOUNDARY> operation;                                   \
    operation.Run(input, weights, inverse, ends, output, descriptor);                                          \
  }
DEFINE_FINALIZE(qwen_owned_finalize_v1, false, false)
DEFINE_FINALIZE(qwen_owned_finalize_half_product_v1, true, false)
DEFINE_FINALIZE(qwen_owned_finalize_fp32_v1, false, true)
