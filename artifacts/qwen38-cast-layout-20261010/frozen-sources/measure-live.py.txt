import argparse
import json
import subprocess
import urllib.request
from pathlib import Path

from tokenizers import Tokenizer

from tools.glm_perf.resident_harness import ResidentClient
from tools.qwen38_decode_study.mixed_prefill_probe import prompt, stream
from tools.qwen4exp.resident_reset import reset_caches
from tools.qwen4exp.thermal_controller import parse_temperatures

ROOT = Path("/srv/ai/src/qwen-cast-layout-20261010")
parser = argparse.ArgumentParser()
parser.add_argument("arm")
args = parser.parse_args()
output = ROOT / (args.arm + "-live.json")
assert not output.exists()
client = ResidentClient("http://127.0.0.1:8001", 6, 180)
model = "/srv/ai/models/Qwen3.8-Flash-Next-W4A16-G128-TP6-MTP-20261010"
tokenizer = Tokenizer.from_file(model + "/tokenizer.json")
short = prompt(tokenizer, 128, "qwen-cast-layout-matched-v17-serial")
long = prompt(tokenizer, 4096, "qwen-cast-layout-matched-v17-cold")
result = {"arm": args.arm, "short": [], "before": client.rpc("resident_status")}


def admission():
    temperatures = parse_temperatures(subprocess.check_output(["npu-smi", "info"], text=True, timeout=5), 6)
    assert max(temperatures) < 85, temperatures
    assert not client.request("/is_paused", method="GET")["is_paused"]
    metrics = urllib.request.urlopen(client.base_url + "/metrics", timeout=10).read().decode()
    assert all(float(line.rsplit(" ", 1)[1]) == 0 for line in metrics.splitlines()
               if line.startswith(("vllm:num_requests_running{", "vllm:num_requests_waiting{")))
    return temperatures


for index in range(3):
    temperatures = admission()
    reset_caches(client)
    value = stream(client.base_url, "qwen38-flash-next", short, 32 if index == 0 else 256, 180)
    assert value["cached_tokens"] == 0
    value["temperatures_before"] = temperatures
    value["warmup"] = index == 0
    result["short"].append(value)
    output.write_text(json.dumps(result, indent=2) + "\n")
    print(json.dumps({"arm": args.arm, "phase": "short", "index": index,
                      "tok_s": value["decode_tokens_per_s"], "ttft_s": value["time_to_first_token_s"]}), flush=True)
temperatures = admission()
reset_caches(client)
result["cold"] = stream(client.base_url, "qwen38-flash-next", long, 16, 180)
assert result["cold"]["cached_tokens"] == 0
result["cold"]["temperatures_before"] = temperatures
result["after"] = client.rpc("resident_status")
assert [r["weight_storage_digest"] for r in result["before"]] == [r["weight_storage_digest"] for r in result["after"]]
assert all(not r["graphs_dirty"] and not r["native_failed"] for r in result["after"])
assert all(s["spill_count"] == 0 and s["restore_count"] == 0
           for r in result["after"] for s in r["prefix_mamba"].values())
output.write_text(json.dumps(result, indent=2) + "\n")
print(json.dumps({"arm": args.arm, "cold_ttft_s": result["cold"]["time_to_first_token_s"]}), flush=True)
