import json,statistics,time
from pathlib import Path
import torch,torch_npu
ROOT=Path('/srv/ai/storage/artifacts/qwen-owned-route-control-20261011')
def prepare_and_qualify():
    scope={'__name__':'qwen_native_route_control'}
    exec(compile((ROOT/'native_route_control.py').read_text(),str(ROOT/'native_route_control.py'),'exec'),scope)
    Kernel=torch.classes.qwen_prefill_v27.Kernel;launch=torch.ops.qwen_prefill_v27.launch
    binary=str(ROOT/'route-control.bin')
    dispatch=scope['NativeFixedRouteDispatch'](Kernel(binary,'qwen_fixed_route_dispatch_v1'),launch)
    finalizers={name:scope['NativeOwnedFinalize'](Kernel(binary,symbol),launch) for name,symbol in (
        ('float_product','qwen_owned_finalize_v1'),('half_product','qwen_owned_finalize_half_product_v1'),('fp32','qwen_owned_finalize_fp32_v1'))}
    from vllm_ascend.models.qwen4_exp.w4_moe import build_grouped_expert_dispatch,finalize_grouped_routes
    device=torch.device('npu',torch.npu.current_device());generator=torch.Generator().manual_seed(2341)
    result={'rank':torch.distributed.get_rank(),'passed':True,'serving_modified':False,'dispatch':[],'finalize':[]}
    with torch.inference_mode():
        for tokens in (1,3,6):
            ids=torch.randint(0,20,(tokens,10),generator=generator).to(device)
            weights=torch.rand(tokens,10,generator=generator).to(device);weights/=weights.sum(1,keepdim=True)
            reference=build_grouped_expert_dispatch(weights,ids,num_local_experts=4,expert_offset=7,weight_dtype=torch.float32,count_mode='histogram')
            expected_order=reference.token_indices.index_select(0,reference.order.int()).int().contiguous()
            actual=dispatch(ids,num_local_experts=4,expert_offset=7)
            torch.npu.synchronize()
            for a,b in zip(actual,(expected_order,reference.inverse_order.int(),reference.group_list)):
                torch.testing.assert_close(a.cpu(),b.cpu(),rtol=0,atol=0)
            timing={'native':[],'torch':[]}
            graphs={}
            for mode in timing:
                graph=torch.npu.NPUGraph()
                with torch.npu.graph(graph):
                    if mode=='native':output=dispatch(ids,num_local_experts=4,expert_offset=7)
                    else:
                        d=build_grouped_expert_dispatch(weights,ids,num_local_experts=4,expert_offset=7,weight_dtype=torch.float32,count_mode='histogram')
                        output=(d.token_indices.index_select(0,d.order.int()).int().contiguous(),d.inverse_order.int(),d.group_list)
                graphs[mode]=(graph,output)
            for repeat in range(4):
                for mode in ('native','torch') if repeat%2==0 else ('torch','native'):
                    torch.npu.synchronize();start=time.perf_counter()
                    for _ in range(60):graphs[mode][0].replay()
                    torch.npu.synchronize();timing[mode].append((time.perf_counter()-start)*1000/60)
            for changed in (torch.full_like(ids,19),torch.full_like(ids,7),ids.roll(1,dims=1)):
                ids.copy_(changed)
                for graph,_ in graphs.values():graph.replay()
                torch.npu.synchronize()
                for a,b in zip(graphs['native'][1],graphs['torch'][1]):torch.testing.assert_close(a.cpu(),b.cpu(),rtol=0,atol=0)
            result['dispatch'].append({'tokens':tokens,'exact':True,'median_ms':{k:statistics.median(v) for k,v in timing.items()}})
            del graphs
        for tokens in (3,6,128):
            ids=torch.randint(0,20,(tokens,10),generator=generator).to(device)
            weights=torch.rand(tokens,10,generator=generator).to(device);weights/=weights.sum(1,keepdim=True)
            ref=build_grouped_expert_dispatch(weights,ids,num_local_experts=4,expert_offset=7,weight_dtype=torch.float32,count_mode='histogram')
            end=ref.group_list;active=int(end[-1].cpu());inverse=ref.inverse_order.int().contiguous()
            routed=(torch.randn(tokens*10,2560,generator=generator)*3).half().to(device);routed[active:]=0
            expected=finalize_grouped_routes(routed,ref,weights,torch.float32,'cann_v2')
            poisoned=routed.clone();poisoned[active:]=float('nan')
            case={'tokens':tokens,'active':active,'methods':{}}
            for mode,operation in finalizers.items():
                actual=operation(poisoned,weights,inverse,end)
                torch.npu.synchronize();a,b=actual.cpu(),expected.cpu()
                case['methods'][mode]={'mismatch':int(torch.count_nonzero(a!=b)),'max_abs':float((a-b).abs().max()),'finite':bool(torch.isfinite(a).all())}
            result['finalize'].append(case)
    (ROOT/('validation-rank-%d.json'%result['rank'])).write_text(json.dumps(result,indent=2)+'\n')
    return result
