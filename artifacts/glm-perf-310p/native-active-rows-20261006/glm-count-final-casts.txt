import csv,json,sys
from pathlib import Path
from decimal import Decimal
root=Path('/srv/ai/artifacts/glm-native-active-rows-20261006/trace-bf16-v5/'+sys.argv[1]+'/rank'+sys.argv[2])
summary=next(root.glob('PROF_*/mindstudio_profiler_output/op_summary*.csv'));timing=next(root.glob('PROF_*/mindstudio_profiler_output/step_trace*.csv'))
steps=[s for s in json.loads((root/'host-steps.json').read_text()) if s['tokens']>0 and 'end_wall_ns' in s]
def ns(v):return int(Decimal(v.strip())*1000)
rows=list(csv.DictReader(summary.open()));records=[]
for step,time in zip(steps,csv.DictReader(timing.open())):
 end=ns(time['Iteration End(us)']);begin=end-ns(time['Iteration Time(us)']);sel=[r for r in rows if begin<=ns(r['Task Start Time(us)'])<end]
 casts=[r for r in sel if r['Task Type']=='AI_CPU' and r['Op Name']=='Cast']
 if step['tokens']<=8:
  records.append({'step':step['step'],'casts':len(casts),'bf16_kernels':sum(r['Op Name']=='glm_bf16_cast_v1' for r in sel),'native_gate_up':sum(r['Op Name']=='glm_fused_gate_up_v1' for r in sel),'native_down':sum(r['Op Name']=='glm_fused_down_v1' for r in sel),'ms':sum(float(r['Task Duration(us)']) for r in casts)/1000})
  if len(records)==1:
   for i,r in enumerate(sel):
    if r['Task Type']=='AI_CPU' and r['Op Name']=='Cast':print('REMAINING',[(x['Op Name'],x['Task Type']) for x in sel[i-3:i+4]])
print(json.dumps(records))
