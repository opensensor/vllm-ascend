import importlib.util,json
from pathlib import Path
import torch,torch_npu

torch.set_num_threads(2);torch.npu.set_device(5);torch.npu.set_compile_mode(jit_compile=False);torch.npu.set_per_process_memory_fraction(.02,5)
s=Path('/workspace/m31-direct-gather-ops.py').read_text()
results=[]
with torch.inference_mode():
 for variant,source in (('scalar',s),('vector',s.replace('seq_len = seq_lens[request]','seq_len = seq_lens[request : request + 1]'))):
  ns={};exec(compile(source,'<attention>', 'exec'),ns);fn=ns['paged_sparse_attention']
  g=torch.Generator().manual_seed(82)
  def random(*shape):return torch.randn(*shape,generator=g).half().to('npu:5')
  q,k,v,iq,ik=random(1,32,128),random(1,2,128),random(1,2,128),random(1,2,128),random(1,128)
  kc,vc,ic=random(4,128,2,128),random(4,128,2,128),random(4,128,128)
  lens=torch.tensor([133],device='npu:5',dtype=torch.int32);slots=torch.tensor([132],device='npu:5');table=torch.arange(4,device='npu:5',dtype=torch.int32)[None]
  args=(q,k,v,iq,ik,kc,vc,ic)

   torch.npu.synchronize();graph.reset()
