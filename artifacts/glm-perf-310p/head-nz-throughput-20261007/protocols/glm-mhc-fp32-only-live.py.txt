import sys,json,uuid
from pathlib import Path
sys.path.insert(0,'/srv/ai/src/glm-selective-w3-nz-test-20261004')
from tools.glm_perf.resident_control import Control
from tools.glm_perf.resident_harness import ResidentClient
version=int(sys.argv[1]);assert version in (56,74)
root=Path('/home/matteius/experiments/glm-reconstruction-20261006');artifacts=Path('/srv/ai/artifacts/glm-native-prefill-20261006')
for rank in range(4):
 audit=json.loads((root/f'mhc-state-flags-rank{rank}.json').read_text())
 fused=[r for r in audit['records'] if r['type']=='MHCFusedPostPreOp'];assert fused and all(r['fp16'] for r in fused)
out=artifacts/f'graph-qualified-v{version}-fp32-mhc-v1';out.mkdir(exist_ok=False)
base=(artifacts/f'graph-qualified-v{version}-v1/candidate.py').read_text().replace('def replacements(native_resources):','def fp32_mhc_base_replacements(native_resources):')
source=base+'''
def replacements(native_resources):
 import torch
 from vllm_ascend.patch.worker import patch_mhc_norm
 changes=fp32_mhc_base_replacements(native_resources)
 target='vllm_ascend.patch.worker.patch_mhc_norm:_mhc_post_torch'
 original=changes.get(target,patch_mhc_norm._mhc_post_torch)
 original=getattr(original,'__glm_native_post_original__',original)
 operation=native_resources['mhc_post_v1']
 selected=operation['wrap_post'](original,operation['native'])
 def post(x,residual,p,comb):
  return selected(x,residual,p,comb) if x.dtype==torch.float32 else original(x,residual,p,comb)
 post.__glm_native_post_original__=original
 changes[target]=post
 return changes
'''
(out/'candidate.py').write_text(source)
c=ResidentClient('http://127.0.0.1:8001',timeout=900)
name=f'native_v{version}_fp32_mhc_prefill_graph'

after=c._acknowledged_rpc('resident_status',matches=lambda rs:c._status_matches(rs) and all(r.get('candidate')==name and r.get('prefill_graph',{}).get('replays',0)>=2 for r in rs))
assert all(not r['graphs_dirty'] and not r['native_failed'] and r['reconstruction']['fallback_dispatches']==0 for r in after)
(out/'qualification.json').write_text(json.dumps(dict(complete=True,response=response,after=after),indent=2));print(name,'qualified',flush=True)
