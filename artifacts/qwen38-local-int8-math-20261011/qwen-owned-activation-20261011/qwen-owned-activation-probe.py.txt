import json,statistics,time
from pathlib import Path
import torch
import torch_npu
ROOT=Path('/srv/ai/storage/artifacts/qwen-owned-activation-20261011')
def prepare_and_qualify():
    scope={'__name__':'qwen_local_swiglu_resources'}
    exec(compile((ROOT/'native_channel_int8.py').read_text(),str(ROOT/'native_channel_int8.py'),'exec'),scope)
    Kernel=torch.classes.qwen_prefill_v27.Kernel;launch=torch.ops.qwen_prefill_v27.launch
    fused=scope['NativeLocalChannelSwiGLU'](Kernel(str(ROOT/'local-channel-swiglu.bin'),'qwen_local_channel_swiglu_v1'),launch)
    binary='/srv/ai/src/qwen-hybrid-prefill-build-20261011-v22/native_hybrid_prefill.bin'
    direct=scope['NativeDirectChannelActivation'](Kernel(binary,'qwen_direct_channel_activation_v1'),launch,
        Kernel(binary,'qwen_direct_local_channel_activation_v1'))
    device=torch.device('npu',torch.npu.current_device());generator=torch.Generator().manual_seed(8152)
    result={'rank':torch.distributed.get_rank(),'serving_modified':False,'cases':[],'exact':True}
    with torch.inference_mode():
        for rows,active in ((30,5),(60,10),(1280,213),(6400,1067)):
            inputs=(torch.randn(rows,1280,generator=generator)*2).half().to(device)
            ends=torch.tensor([0,active//2,active],dtype=torch.int64,device=device)
            expected=direct(torch_npu.npu_swiglu(inputs,dim=-1),ends);actual=fused(inputs,ends)
            torch.npu.synchronize()
            mismatch=int(torch.count_nonzero(actual[0][:active].cpu()!=expected[0][:active].cpu()))
            scale_difference=float((actual[1][:active].cpu()-expected[1][:active].cpu()).abs().max())
            timing={'fused':[],'separate':[]}
            graphs={}
            for name in timing:
                graph=torch.npu.NPUGraph()
                with torch.npu.graph(graph):
                    output=fused(inputs,ends) if name=='fused' else direct(torch_npu.npu_swiglu(inputs,dim=-1),ends)
                graphs[name]=(graph,output)
            for repeat in range(4):
                for name in ('fused','separate') if repeat%2==0 else ('separate','fused'):
                    torch.npu.synchronize();start=time.perf_counter()
                    for _ in range(40):graphs[name][0].replay()
                    torch.npu.synchronize();timing[name].append((time.perf_counter()-start)*1000/40)
            replay_mismatch=0
            for count in (0,1,active,0,rows):
                ends.copy_(torch.tensor([0,count//2,count],dtype=torch.int64,device=device))
                for graph,_ in graphs.values():graph.replay()
                torch.npu.synchronize()
                replay_mismatch+=int(torch.count_nonzero(graphs['fused'][1][0][:count].cpu()!=graphs['separate'][1][0][:count].cpu()))
            result['exact'] &= mismatch==0 and scale_difference==0 and replay_mismatch==0
            result['cases'].append({'rows':rows,'active':active,'mismatched_codes':mismatch,'scale_max_abs':scale_difference,
                'replay_mismatched_codes':replay_mismatch,'median_ms':{k:statistics.median(v) for k,v in timing.items()}})
            del graphs
    result['passed']=True
    (ROOT/('validation-rank-%d.json'%result['rank'])).write_text(json.dumps(result,indent=2)+'\n')
    return result
